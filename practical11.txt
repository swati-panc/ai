# PRACTICAL 11: TRANSFORMER DEMO USING PRETRAINED MODEL

# Step 1: Import libraries
from transformers import pipeline, AutoTokenizer, AutoModelForSequenceClassification
import torch

print("Libraries imported successfully!")
print("PyTorch Version:", torch.__version__)

# Check device
device = 0 if torch.cuda.is_available() else -1
print("Device:", "CUDA (GPU)" if device == 0 else "CPU")


# Step 2: Load pretrained Transformer model
print("\nLoading Pretrained Transformer...")

sentiment = pipeline(
    "sentiment-analysis",
    model="distilbert-base-uncased-finetuned-sst-2-english",
    device=device
)

print("Model loaded successfully!")


# Step 3: Sentiment Classification
sentences = [
    "I absolutely love learning Natural Language Processing!",
    "This was the most boring and terrible movie I have ever watched.",
    "Transformers with self-attention are truly revolutionary.",
    "The customer service was disappointing and very slow.",
    "The food was delicious and the ambiance was fantastic."
]

print("\nSentiment Classification:")

for i, text in enumerate(sentences, 1):
    result = sentiment(text)[0]

    print("\nSample", i)
    print("Text       :", text)
    print("Prediction :", result["label"])
    print("Confidence :", round(result["score"], 4))


# Step 4: Tokenization
print("\n\nUnder The Hood: Tokenization")

text = "Transformers are amazing for NLP tasks!"

tokenizer = AutoTokenizer.from_pretrained(
    "distilbert-base-uncased-finetuned-sst-2-english"
)

tokens = tokenizer.tokenize(text)
inputs = tokenizer(text, return_tensors="pt")

print("Original Text :", text)
print("Tokens        :", tokens)
print("Input IDs     :", inputs["input_ids"].tolist())
print("Attention Mask:", inputs["attention_mask"].tolist())


# Step 5: Manual Forward Pass
print("\n\nManual Forward Pass")

model = AutoModelForSequenceClassification.from_pretrained(
    "distilbert-base-uncased-finetuned-sst-2-english"
)

with torch.no_grad():
    output = model(**inputs)

logits = output.logits
probabilities = torch.softmax(logits, dim=1)

predicted_class = torch.argmax(probabilities, dim=1).item()

labels = ["NEGATIVE", "POSITIVE"]

print("Raw Logits:", logits.tolist()[0])
print("Negative Probability:", round(probabilities[0][0].item(), 4))
print("Positive Probability:", round(probabilities[0][1].item(), 4))
print("Predicted Label:", labels[predicted_class])


# Step 6: Text Generation using DistilGPT-2
print("\n\nText Generation")

generator = pipeline(
    "text-generation",
    model="distilgpt2"
)

prompt = "Natural language processing with transformers is"

result = generator(
    prompt,
    max_length=40,
    num_return_sequences=1
)

print("Prompt:", prompt)
print("Generated Text:", result[0]["generated_text"])


# Step 7: Custom Sentence
print("\n\nCustom Sentence Testing")

custom_text = "The practical experiment was intuitive, powerful, and easy to follow."

result = sentiment(custom_text)[0]

print("Custom Input:", custom_text)
print("Predicted:", result["label"])
print("Confidence:", round(result["score"], 4))
